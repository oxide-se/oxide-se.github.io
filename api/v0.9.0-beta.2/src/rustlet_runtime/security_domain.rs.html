20/// Minimal lifecycle state carried by one Security Domain instance.
21#[derive(Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
22#[serde(crate = "crate::serde")]
23pub enum SecurityDomainLifecycle {
24    /// Installed but not currently authorized to exercise management.
25    Installed,
26    /// Selectable lifecycle; management still requires the relevant privileges.
27    Selectable,
28    /// Locked lifecycle; management hooks reject by default.
29    Locked,
30}
31
32impl SecurityDomainLifecycle {
33    const INSTALLED_BYTE: u8 = 0x03;
34    const SELECTABLE_BYTE: u8 = 0x07;
35    const LOCKED_BYTE: u8 = 0x7f;
36
37    /// Returns the encoded lifecycle byte used by `GET DATA 9F70`.
38    pub const fn as_byte(self) -> u8 {
39        match self {
40            Self::Installed => Self::INSTALLED_BYTE,
41            Self::Selectable => Self::SELECTABLE_BYTE,
42            Self::Locked => Self::LOCKED_BYTE,
43        }
44    }
45
46    /// Decodes one stored lifecycle byte.
47    ///
48    /// `0x03` maps to Installed and `0x7F` to Locked. Every other byte maps
49    /// to Selectable, including the legacy `0x00`; this is not a strict validator.
50    pub const fn from_stored_byte(byte: u8) -> Self {
51        match byte {
52            Self::INSTALLED_BYTE => Self::Installed,
53            Self::LOCKED_BYTE => Self::Locked,
54            _ => Self::Selectable,
55        }
56    }
57
58    /// Returns true if the instance may currently exercise management authority.
59    pub const fn may_manage(self) -> bool {
60        matches!(self, Self::Selectable)
61    }
62}

72impl SecurityDomainAdministrativeState {
73    const LIFECYCLE_RESPONSE_LEN: usize = 4;
74
75    /// Creates an empty administrative state.
76    pub const fn empty() -> Self {
77        Self {
78            privileges: SecurityDomainPrivileges::empty(),
79            lifecycle: SecurityDomainLifecycle::Selectable,
80        }
81    }
82
83    /// Creates an administrative state from decoded privileges and lifecycle.
84    pub const fn from_parts(
85        privileges: SecurityDomainPrivileges,
86        lifecycle: SecurityDomainLifecycle,
87    ) -> Self {
88        Self {
89            privileges,
90            lifecycle,
91        }
92    }
93
94    /// Returns the effective Security Domain privileges.
95    pub const fn privileges(self) -> SecurityDomainPrivileges {
96        self.privileges
97    }
98
99    /// Returns the current lifecycle.
100    pub const fn lifecycle(self) -> SecurityDomainLifecycle {
101        self.lifecycle
102    }
103
104    /// Writes the standard `GET DATA 9F70` response for this instance.
105    pub fn write_lifecycle_data(self, out: &mut [u8]) -> Result<usize, ApduStatus> {
106        if out.len() < Self::LIFECYCLE_RESPONSE_LEN {
107            return Err(ApduStatus::wrong_length());
108        }
109        out[0] = 0x9f;
110        out[1] = 0x70;
111        out[2] = 0x01;
112        out[3] = self.lifecycle().as_byte();
113        Ok(Self::LIFECYCLE_RESPONSE_LEN)
114    }
115}

144    /// Creates a permissive development policy that bypasses this value's bit
145    /// checks. It does not bypass kernel ownership/authority validation and is
146    /// not a production authorization policy.

148        Self {
149            bytes: [0xff; 3],
150            encoded_len: 3,

155    /// Decodes the raw GlobalPlatform privilege bytes carried by `INSTALL`.
156    ///
157    /// Accepts zero through three bytes, copying them into inline storage with
158    /// zero padding. More than three returns `TooLong`. Queries interpret the
159    /// first byte and selected second-byte bits; other bits are preserved.
160    /// Decoding is not an authorization decision.
161    pub fn from_install_bytes(bytes: &[u8]) -> Result<Self, SecurityDomainPrivilegeError> {
162        if bytes.len() > 3 {
163            return Err(SecurityDomainPrivilegeError::TooLong);
164        }
165
166        let mut encoded = [0u8; 3];
167        let mut index = 0;
168        while index < bytes.len() {
169            encoded[index] = bytes[index];
170            index += 1;
171        }
172
173        Ok(Self {
174            bytes: encoded,
175            encoded_len: bytes.len() as u8,
176            developer_open: false,
177        })
178    }
179
180    /// Returns the raw encoded privilege bytes.
181    pub const fn encoded_bytes(self) -> [u8; 3] {
182        self.bytes
183    }
184
185    /// Returns how many privilege bytes were explicitly encoded.
186    pub const fn encoded_len(self) -> usize {
187        self.encoded_len as usize
188    }
189
190    /// Returns the first GlobalPlatform privilege byte.
191    pub const fn first_byte(self) -> u8 {
192        self.bytes[0]
193    }

304    /// Returns true if `self` grants every currently implemented privilege bit
305    /// requested by `other`.
306    ///
307    /// The current comparison interprets the whole first GlobalPlatform
308    /// privilege byte and compares extension bytes conservatively as raw bits.
309    pub const fn contains(self, other: Self) -> bool {
310        if self.developer_open {
311            return true;
312        }
313
314        let lhs = self.encoded_bytes();
315        let rhs = other.encoded_bytes();
316        let mut index = 0;
317        while index < 3 {
318            if (lhs[index] & rhs[index]) != rhs[index] {
319                return false;
320            }
321            index += 1;
322        }
323        true
324    }

386/// Secure channel level requested or active for a Security Domain session.
387#[derive(Clone, Copy, PartialEq, Eq)]
388pub struct SecurityLevel {
389    bits: u8,
390}
391
392impl SecurityLevel {
393    /// No command or response protection bits.
394    pub const NONE: Self = Self { bits: 0x00 };
395    /// Command authentication requested.
396    pub const C_MAC: Self = Self { bits: 0x01 };
397    /// Command encryption requested; protocol rules may also require C-MAC.
398    pub const C_ENC: Self = Self { bits: 0x02 };
399    /// Response authentication bit; representation does not imply implementation.
400    pub const R_MAC: Self = Self { bits: 0x10 };
401    /// Response encryption bit; representation does not imply implementation.
402    pub const R_ENC: Self = Self { bits: 0x20 };
403
404    /// Preserve raw GP-style bits without validating reserved or unsupported
405    /// combinations. Protocol establishment must perform that validation.
406    pub const fn from_bits(bits: u8) -> Self {
407        Self { bits }
408    }
409
410    /// Returns the raw bit representation.
411    pub const fn bits(self) -> u8 {
412        self.bits
413    }
414
415    /// Returns true if every bit from `other` is present.
416    pub const fn contains(self, other: Self) -> bool {
417        (self.bits & other.bits) == other.bits
418    }
419}

579/// Management and secure-channel interface implemented by Security Domain Rustlets.
580///
581/// A Security Domain Rustlet remains a normal selectable Rustlet through
582/// `Rustlet::process_apdu`, but it also exposes management and secure-channel
583/// hooks that the kernel can call through the dedicated Security Domain proxy.
584/// Administrative privilege is not privileged CPU execution. The kernel keeps
585/// ownership, hierarchy and non-escalation checks; returning `Ok` here cannot
586/// grant authority beyond those checks. Authorization hooks do not themselves
587/// commit registry mutations.
588///
589/// Request slices and output buffers are loans for this invocation only. Do not
590/// retain them in instance state. Output hooks write into `out` and return the
591/// initialized prefix length, at most `out.len()`; on error, output must not be
592/// published. In-place unwrap/wrap is an ABI optimization handled by the proxy,
593/// not permission to construct overlapping mutable Rust references.
594///
595/// Defaults deny unimplemented protocol operations with `6D00`, advertise no
596/// protocol, and report no open channel. Management defaults require a selectable
597/// lifecycle and the relevant privilege and otherwise return `6985`.
598/// `get_data` supplies discovery/lifecycle objects and reports missing or
599/// unsupported tags. Override protocol hooks and capability queries together.
600///
601/// With `declare_security_domain!`, the persistent wrapper owns administrative
602/// state. Its active secure-channel state can remain resident between calls;
603/// exclude session secrets from serde and erase them in `reset_secure_channel`.
604/// SCP03 key loading is available only during kernel-initiated SDDISPATCH.
605///
606/// # API map
607///
608/// - Management policy: [`Self::install_for_load`], [`Self::install_for_install`],
609///   [`Self::delete_aid`], [`Self::put_key`], [`Self::store_data`],
610///   [`Self::set_status`] and [`Self::get_data`].
611/// - Establishment: [`Self::handle_delegated_secure_channel_command`],
612///   [`Self::initialize_update`], [`Self::external_authenticate`] and the
613///   `scp11_*` hooks below. Advertise only implemented protocol capabilities.
614/// - Secure messaging: [`Self::unwrap_command`], [`Self::wrap_response`],
615///   [`Self::current_security_level`] and [`Self::reset_secure_channel`].
616/// - Kernel key service: [`crate::RuntimeSyscall::SecurityDomainLoadScp03Key`],
617///   [`crate::Scp03LoadKeyParams`] and
618///   [`crate::syscall::runtime::crypto::load_scp03_key::trigger`]. This service
619///   is denied during ordinary `process_apdu` and installation, even for an SD.
620/// - Shared cryptography: [`crate::CryptoProvider`], [`crate::Cipher`],
621///   [`crate::Mac`] and [`crate::KeyAgreement`].
622///
623/// # Current rollback limitation
624///
625/// A normally rejected management operation restores its staged registry
626/// changes, but does not yet restore the persistent fields of the resident
627/// SD object. A later call can serialize those abandoned changes again.
628/// Do not assume an error return restores `self`. Session serialization and
629/// separate persistent/session restoration are not implemented yet.
630pub trait RustletSecurityDomain: Rustlet {
631    /// Returns the standard administrative state carried by this instance.
632    fn administrative_state(&self) -> SecurityDomainAdministrativeState {
633        SecurityDomainAdministrativeState::empty()

703    /// Writes `GET DATA` response bytes into `out`.
704    fn get_data(&mut self, tag: GetDataTag, out: &mut [u8]) -> Result<usize, ApduStatus> {
705        // Invariant: GP discovery objects are derived from the same protocol
706        // hooks used by establishment dispatch, so a Rustlet SD cannot
707        // accidentally advertise a kernel-only protocol.
708        let capabilities = crate::gp::SecurityDomainCapabilities {
709            scp03_s8: self.supports_scp03_s8(),
710            scp03_s16: self.supports_scp03_s16(),
711            scp11a: self.supports_scp11a(),
712            scp11b: self.supports_scp11b(),
713            scp11c: self.supports_scp11c(),
714        };
715        match tag.0 {
716            0x0066 => crate::gp::write_card_recognition_data(capabilities, out)
717                .map_err(|_| ApduStatus::wrong_length()),
718            0x0067 if capabilities.has_secure_channel() => {
719                crate::gp::write_card_capability_information(capabilities, out)
720                    .map_err(|_| ApduStatus::wrong_length())
721            }
722            0x0067 => Err(ApduStatus::referenced_data_not_found()),
723            0x9f70 => self.administrative_state().write_lifecycle_data(out),
724            _ => Err(ApduStatus::instruction_not_supported()),
725        }
726    }

797    /// Advances one establishment protocol owned entirely by this Security
798    /// Domain rather than by the kernel profile matrix.
799    ///
800    /// The default implementation adapts SCP03 to the existing family APIs.
801    /// A future supplementary Security Domain may override this method to
802    /// implement another GP-compatible establishment sequence.
803    fn handle_delegated_secure_channel_command(
804        &mut self,
805        ctx: &mut RustletCtx,
806        command: &DelegatedSecureChannelCommand<'_>,
807        out: &mut [u8],
808    ) -> Result<usize, ApduStatus> {
809        match command.header.ins {
810            0x50 => self.initialize_update(
811                ctx,
812                &InitializeUpdate {
813                    key_version: command.header.p1,
814                    key_id: command.header.p2,
815                    host_challenge: command.data,
816                },
817                out,
818            ),
819            0x82 if command.header.p2 == 0x00 => {
820                self.external_authenticate(
821                    ctx,
822                    &ExternalAuthenticate {
823                        cla: command.header.cla,
824                        security_level: SecurityLevel::from_bits(command.header.p1),
825                        p2: command.header.p2,
826                        authentication_data: command.data,
827                    },
828                )?;
829                Ok(0)
830            }
831            _ => Err(ApduStatus::instruction_not_supported()),
832        }
833    }
