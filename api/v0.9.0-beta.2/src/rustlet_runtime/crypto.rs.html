117impl CryptoError {
118    /// Decode the low byte of a kernel error code; unknown values map to Unsupported.
119    /// Strip the ABI error flag before calling; this does not decode success values.
120    pub const fn from_code(code: usize) -> Self {
121        match code as u8 {
122            value if value == CryptoErrorCode::InvalidKeyLength as u8 => Self::InvalidKeyLength,
123            value if value == CryptoErrorCode::InvalidBufferLength as u8 => {
124                Self::InvalidBufferLength
125            }
126            value if value == CryptoErrorCode::InvalidOutputLength as u8 => {
127                Self::InvalidOutputLength
128            }
129            value if value == CryptoErrorCode::PermissionDenied as u8 => Self::PermissionDenied,
130            value if value == CryptoErrorCode::NotInitialized as u8 => Self::NotInitialized,
131            value if value == CryptoErrorCode::NotFound as u8 => Self::NotFound,
132            _ => Self::Unsupported,
133        }
134    }
135}

137/// Backend used by typed cipher sessions.
138///
139/// Runtime code normally uses the internal syscall backend. Tests may provide a
140/// local backend to exercise the typestate API without entering the kernel.
141pub trait CryptoBackend {
142    /// Execute one complete operation with borrowed, validated cipher parameters.
143    ///
144    /// Key/IV loans last through this synchronous call only. Implementations must
145    /// not retain their addresses. Safe implementations cannot store the borrowed
146    /// references in a longer-lived backend. Return the produced prefix length
147    /// or a service error; do not publish output after an error.
148    fn cipher_do_final(
149        &mut self,
150        parameters: &CipherReady<'_>,
151        input: &[u8],
152        output: &mut [u8],
153    ) -> Result<usize, CryptoError>;
154
155    /// Compute one complete MAC operation.
156    fn mac_compute(
157        &mut self,
158        key: &[u8],
159        algorithm: MacAlgorithm,
160        input: &[u8],
161        output: &mut [u8; MAC_TAG_SIZE],
162    ) -> Result<(), CryptoError>;
163
164    /// Verify one complete MAC operation using kernel-side comparison.
165    fn mac_verify(
166        &mut self,
167        key: &[u8],
168        algorithm: MacAlgorithm,
169        input: &[u8],
170        expected_tag: &[u8],
171    ) -> Result<bool, CryptoError>;
172}

194    /// Writes into `output` and returns the produced byte count. Inputs are

206    /// Compute a MAC over one already assembled message.
207    ///
208    /// This one-shot form avoids the 256-byte accumulation buffer carried by
209    /// [`Mac`]. It is useful when a Rustlet already owns a contiguous message,
210    /// especially on targets with small application stacks. AES-CMAC accepts

362impl CryptoBackend for SyscallCryptoBackend {
363    fn cipher_do_final(
364        &mut self,
365        parameters: &CipherReady<'_>,
366        input: &[u8],
367        output: &mut [u8],
368    ) -> Result<usize, CryptoError> {
369        // Raw addresses exist only in this synchronous ABI record. The borrowed
370        // state keeps key/IV storage alive and immutable throughout the SVC.
371        let params = CryptoCipherDoFinalParams {
372            algorithm: parameters.algorithm as u8,
373            mode: parameters.mode as u8,
374            key_ptr: parameters.key.as_ptr(),
375            key_len: parameters.key.len(),
376            iv_ptr: parameters.iv.as_ptr(),
377            iv_len: parameters.iv.len(),
378            input_ptr: input.as_ptr(),
379            input_len: input.len(),
380            output_ptr: output.as_mut_ptr(),
381            output_capacity: output.len(),
382        };
383        let result = crate::syscall::runtime::crypto::cipher_do_final::trigger(&params);
384        if result & CRYPTO_RESULT_ERROR_FLAG == 0 {
385            Ok(result)
386        } else {
387            Err(CryptoError::from_code(result & !CRYPTO_RESULT_ERROR_FLAG))
388        }
389    }
390
391    fn mac_compute(
392        &mut self,
393        key: &[u8],
394        algorithm: MacAlgorithm,
395        input: &[u8],
396        output: &mut [u8; MAC_TAG_SIZE],
397    ) -> Result<(), CryptoError> {
398        if !mac_key_len_matches_algorithm(key.len(), algorithm) {
399            return Err(CryptoError::InvalidKeyLength);
400        }
401
402        let params = CryptoMacDoFinalParams {
403            algorithm: algorithm as u8,
404            operation: CryptoMacOperation::Compute as u8,
405            key_ptr: key.as_ptr(),
406            key_len: key.len(),
407            input_ptr: input.as_ptr(),
408            input_len: input.len(),
409            expected_tag_ptr: core::ptr::null(),
410            expected_tag_len: 0,
411            output_ptr: output.as_mut_ptr(),
412            output_capacity: output.len(),
413        };
414        let result = crate::syscall::runtime::crypto::mac_do_final::trigger(&params);
415        if result & CRYPTO_RESULT_ERROR_FLAG == 0 {
416            if result == MAC_TAG_SIZE {
417                Ok(())
418            } else {
419                Err(CryptoError::InvalidOutputLength)
420            }
421        } else {
422            Err(CryptoError::from_code(result & !CRYPTO_RESULT_ERROR_FLAG))
423        }
424    }
425
426    fn mac_verify(
427        &mut self,
428        key: &[u8],
429        algorithm: MacAlgorithm,
430        input: &[u8],
431        expected_tag: &[u8],
432    ) -> Result<bool, CryptoError> {
433        if !mac_key_len_matches_algorithm(key.len(), algorithm) {
434            return Err(CryptoError::InvalidKeyLength);
435        }
436
437        let params = CryptoMacDoFinalParams {
438            algorithm: algorithm as u8,
439            operation: CryptoMacOperation::Verify as u8,
440            key_ptr: key.as_ptr(),
441            key_len: key.len(),
442            input_ptr: input.as_ptr(),
443            input_len: input.len(),
444            expected_tag_ptr: expected_tag.as_ptr(),
445            expected_tag_len: expected_tag.len(),
446            output_ptr: core::ptr::null_mut(),
447            output_capacity: 0,
448        };
449        let result = crate::syscall::runtime::crypto::mac_do_final::trigger(&params);
450        if result & CRYPTO_RESULT_ERROR_FLAG == 0 {
451            Ok(result != 0)
452        } else {
453            Err(CryptoError::from_code(result & !CRYPTO_RESULT_ERROR_FLAG))
454        }
455    }
456}

458fn key_len_matches_algorithm(len: usize, algorithm: Algorithm) -> bool {
459    match algorithm {
460        Algorithm::Aes128CbcNoPadding | Algorithm::Aes128CbcIso9797M2 => len == AES128_KEY_SIZE,
461        Algorithm::Aes256CbcNoPadding | Algorithm::Aes256CbcIso9797M2 => len == AES256_KEY_SIZE,
462        Algorithm::Aes128EcbNoPadding => len == AES128_KEY_SIZE,
463        Algorithm::Aes256EcbNoPadding => len == AES256_KEY_SIZE,
464    }
465}
466
467fn mac_key_len_matches_algorithm(len: usize, algorithm: MacAlgorithm) -> bool {
468    match algorithm {
469        MacAlgorithm::AesCmac => len == AES128_KEY_SIZE || len == AES256_KEY_SIZE,
470    }
471}

505/// Validated cipher state borrowing key and IV without copying.
506///
507/// Created by [`Cipher::init`]. The key length matches the algorithm; the IV
508/// is exactly 16 bytes, including for ECB to preserve the current ABI contract.
509/// This state carries actual Rust references. A backend only borrows the record
510/// for one atomic operation.
511///
512/// A backend cannot extend an input loan into its own longer-lived storage:
513///
514/// ```compile_fail
515/// use rustlet_runtime::CipherReady;
516/// fn retain<'backend>(saved: &mut Option<&'backend [u8]>, params: &CipherReady<'_>) {
517///     *saved = Some(params.key());
518/// }
519/// ```
520pub struct CipherReady<'key> {
521    key: &'key [u8],
522    iv: &'key [u8; AES_BLOCK_SIZE],
523    mode: CipherMode,
524    algorithm: Algorithm,
525}

549/// Typestate cipher session with exclusive access to its backend.
550///
551/// `Cipher<Uninitialized>` contains no key/IV storage. Initialization consumes
552/// it and produces `Cipher<CipherReady<'key>>`, borrowing those bytes through
553/// finalization or abandonment. Backend and key lifetimes are independent, so
554/// a long-lived provider can process successive short-lived keys.
555///
556/// No key/IV copy, heap allocation or runtime initialization flag is needed.
557/// [`Cipher::finish`] consumes the ready session on both success and error.
558/// Dropping an unfinished session releases its loans without a kernel call.
559pub struct Cipher<'backend, State> {
560    backend: &'backend mut dyn CryptoBackend,
561    state: State,
562}
563
564impl<'backend> Cipher<'backend, Uninitialized> {
565    /// Exclusively borrow a backend for an unconfigured session.
566    #[inline(always)]
567    pub fn new(backend: &'backend mut dyn CryptoBackend) -> Self {
568        Self {
569            backend,
570            state: Uninitialized,
571        }
572    }
573
574    /// Validate and borrow a key/IV, consuming the uninitialized session.
575    ///
576    /// The IV must be 16 bytes even for ECB (otherwise InvalidBufferLength);
577    /// the key must match the algorithm (otherwise InvalidKeyLength).
578    /// Validation makes no backend call. Both arrays remain immutably borrowed
579    /// while the returned session can still be used.
580    ///
581    /// The compiler rejects expired or mutated key material and reuse of a
582    /// consumed session. These examples compile-check the actual public API:
583    ///
584    /// ```compile_fail,E0597
585    /// use rustlet_runtime::{Algorithm, Cipher, CipherMode, RustletCtx};
586    /// let mut ctx = RustletCtx::new();
587    /// let mut provider = ctx.crypto();
588    /// let iv = [0; 16];
589    /// let ready;
590    /// {
591    ///     let key = [0; 16];
592    ///     ready = Cipher::new(&mut provider)
593    ///     .init(&key, &iv, CipherMode::Encrypt, Algorithm::Aes128CbcNoPadding)
594    ///     .unwrap();
595    /// }
596    /// ready.finish(&[0; 16], &mut [0; 16]);
597    /// ```
598    ///
599    /// ```compile_fail,E0597
600    /// use rustlet_runtime::{Algorithm, Cipher, CipherMode, RustletCtx};
601    /// let mut ctx = RustletCtx::new();
602    /// let mut provider = ctx.crypto();
603    /// let key = [0; 16];
604    /// let ready;
605    /// {
606    ///     let iv = [0; 16];
607    ///     ready = Cipher::new(&mut provider)
608    ///     .init(&key, &iv, CipherMode::Encrypt, Algorithm::Aes128CbcNoPadding)
609    ///     .unwrap();
610    /// }
611    /// ready.finish(&[0; 16], &mut [0; 16]);
612    /// ```
613    ///
614    /// ```compile_fail,E0506
615    /// use rustlet_runtime::{Algorithm, Cipher, CipherMode, RustletCtx};
616    /// let mut ctx = RustletCtx::new();
617    /// let mut provider = ctx.crypto();
618    /// let mut key = [0; 16];
619    /// let iv = [0; 16];
620    /// let ready = Cipher::new(&mut provider)
621    ///     .init(&key, &iv, CipherMode::Encrypt, Algorithm::Aes128CbcNoPadding)
622    ///     .unwrap();
623    /// key[0] = 1;
624    /// ready.finish(&[0; 16], &mut [0; 16]);
625    /// ```
626    ///
627    /// ```compile_fail,E0506
628    /// use rustlet_runtime::{Algorithm, Cipher, CipherMode, RustletCtx};
629    /// let mut ctx = RustletCtx::new();
630    /// let mut provider = ctx.crypto();
631    /// let key = [0; 16];
632    /// let mut iv = [0; 16];

635    ///     .unwrap();
636    /// iv[0] = 1;
637    /// ready.finish(&[0; 16], &mut [0; 16]);
638    /// ```
639    ///
640    /// ```compile_fail,E0382
641    /// use rustlet_runtime::{Algorithm, Cipher, CipherMode, RustletCtx};
642    /// let mut ctx = RustletCtx::new();
643    /// let mut provider = ctx.crypto();
644    /// let key = [0; 16];
645    /// let iv = [0; 16];
646    /// let ready = Cipher::new(&mut provider)
647    ///     .init(&key, &iv, CipherMode::Encrypt, Algorithm::Aes128CbcNoPadding)
648    ///     .unwrap();
649    /// ready.finish(&[0; 16], &mut [0; 16]);
650    /// ready.finish(&[0; 16], &mut [0; 16]);
651    /// ```
652    ///
653    /// ```compile_fail,E0599
654    /// use rustlet_runtime::{Algorithm, Cipher, CipherMode, RustletCtx};
655    /// let mut ctx = RustletCtx::new();
656    /// let mut provider = ctx.crypto();
657    /// Cipher::new(&mut provider).finish(&[0; 16], &mut [0; 16]);
658    /// ```
659    // Keep the borrowed transition in its caller so the optimizer can eliminate
660    // aggregate Result/session temporaries on ARMv6-M.
661    #[inline(always)]
662    pub fn init<'key>(
663        self,
664        key: &'key [u8],
665        iv: &'key [u8],
666        mode: CipherMode,
667        algorithm: Algorithm,
668    ) -> Result<Cipher<'backend, CipherReady<'key>>, CryptoError> {
669        let iv = iv
670            .try_into()
671            .map_err(|_| CryptoError::InvalidBufferLength)?;
672        if !key_len_matches_algorithm(key.len(), algorithm) {
673            return Err(CryptoError::InvalidKeyLength);
674        }
675        Ok(Cipher {
676            backend: self.backend,
677            state: CipherReady {
678                key,
679                iv,
680                mode,
681                algorithm,
682            },
683        })
684    }
685}

687/// Typestate MAC session.
688///
689/// Data is accumulated Rustlet-side with `update`. The kernel sees only one
690/// final MAC syscall when `compute` or `verify` consumes the ready session.
691/// Owns a 256-byte message buffer and a 32-byte key buffer inline; moving the
692/// typestate may copy them. Use `CryptoProvider::compute_mac` when the message
693/// already exists. There is no heap allocation or automatic zeroizing Drop.
694pub struct Mac<'a, State> {
695    backend: &'a mut dyn CryptoBackend,
696    key: [u8; AES256_KEY_SIZE],
697    key_len: usize,
698    algorithm: MacAlgorithm,
699    input: [u8; MAC_INPUT_CAPACITY],
700    input_len: usize,
701    _state: PhantomData<State>,
702}
703
704impl<'a> Mac<'a, Uninitialized> {
705    /// Create a new unconfigured MAC session.
706    pub fn new(backend: &'a mut dyn CryptoBackend) -> Self {
707        Self {
708            backend,
709            key: [0; AES256_KEY_SIZE],
710            key_len: 0,
711            algorithm: MacAlgorithm::AesCmac,
712            input: [0; MAC_INPUT_CAPACITY],
713            input_len: 0,
714            _state: PhantomData,
715        }
716    }
717
718    /// Copy a 16- or 32-byte key into this session and reset its message length.
719    /// Other key lengths return `InvalidKeyLength`; the key borrow then ends.
720    pub fn init(
721        mut self,
722        key: &[u8],
723        algorithm: MacAlgorithm,
724    ) -> Result<Mac<'a, Ready>, CryptoError> {
725        if !mac_key_len_matches_algorithm(key.len(), algorithm) {
726            return Err(CryptoError::InvalidKeyLength);
727        }
728
729        self.key[..key.len()].copy_from_slice(key);
730        Ok(Mac {
731            backend: self.backend,
732            key: self.key,
733            key_len: key.len(),
734            algorithm,
735            input: self.input,
736            input_len: 0,
737            _state: PhantomData,
738        })
739    }
740}
741
742impl Mac<'_, Ready> {
743    /// Copy bytes into the inline message accumulator.
744    /// More than 256 total bytes returns `InvalidBufferLength` without changing
745    /// the accumulated prefix. This does not call a streaming kernel service.
746    pub fn update(&mut self, data: &[u8]) -> Result<(), CryptoError> {
747        let Some(end) = self.input_len.checked_add(data.len()) else {
748            return Err(CryptoError::InvalidBufferLength);
749        };
750        if end > self.input.len() {
751            return Err(CryptoError::InvalidBufferLength);
752        }
753
754        self.input[self.input_len..end].copy_from_slice(data);
755        self.input_len = end;
756        Ok(())
757    }
758
759    /// Finalize the MAC and return the computed tag.
760    pub fn compute(self) -> Result<MacTag, CryptoError> {
761        let mut tag = [0u8; MAC_TAG_SIZE];
762        self.backend.mac_compute(
763            &self.key[..self.key_len],
764            self.algorithm,
765            &self.input[..self.input_len],
766            &mut tag,
767        )?;
768        Ok(MacTag(tag))
769    }
770
771    /// Finalize the MAC and compare the tag in the kernel.
772    /// The expected tag must be the complete 16-byte AES-CMAC tag.
773    /// `Ok(false)` is an authentication mismatch, not a service error.
774    /// Invalid tag lengths and kernel failures return `Err`; the session is
775    /// consumed on every outcome.
776    pub fn verify(self, expected_tag: &[u8]) -> Result<bool, CryptoError> {
777        self.backend.mac_verify(
778            &self.key[..self.key_len],
779            self.algorithm,
780            &self.input[..self.input_len],
781            expected_tag,
782        )
783    }
784}
785
786impl Cipher<'_, CipherReady<'_>> {
787    /// Always return `Unsupported` without reading input or changing output.
788    /// The ready session remains available for `finish`; streaming is not implemented.
789    pub fn update(&mut self, _input: &[u8], _output: &mut [u8]) -> Result<usize, CryptoError> {
790        Err(CryptoError::Unsupported)
791    }
792
793    /// Complete one cipher operation and consume the ready session on any outcome.
794    /// Return the produced prefix length. No-padding input must be block aligned;
795    /// padded encryption needs room for the added block/padding. Kernel errors
796    /// include invalid input/padding, insufficient output and denied ranges.
797    /// Do not use partially written output on failure.
798    #[inline(always)]
799    pub fn finish(self, input: &[u8], output: &mut [u8]) -> Result<usize, CryptoError> {
800        self.backend.cipher_do_final(&self.state, input, output)
801    }
802}

892/// Java Card-style key agreement object backed by one atomic kernel syscall.
893pub struct KeyAgreement<'a, State> {
894    crypto: &'a mut CryptoProvider,
895    curve: EcCurve,
896    private_key: [u8; P256_PRIVATE_KEY_SIZE],
897    _state: PhantomData<State>,
898}
899
900impl<'a> KeyAgreement<'a, Uninitialized> {
901    /// Create a new unconfigured key agreement object.
902    pub fn new(crypto: &'a mut CryptoProvider) -> Self {
903        Self {
904            crypto,
905            curve: EcCurve::P256,
906            private_key: [0; P256_PRIVATE_KEY_SIZE],
907            _state: PhantomData,
908        }
909    }
910
911    /// Copy the 32-byte scalar into this object without validating it yet.
912    /// The input borrow ends immediately. Invalid scalars are rejected by the
913    /// kernel when an agreement is performed.
914    pub fn init(
915        mut self,
916        private_key: &EcPrivateKey,
917    ) -> Result<KeyAgreement<'a, Ready>, CryptoError> {
918        self.curve = private_key.curve();
919        self.private_key.copy_from_slice(private_key.as_bytes());
920        Ok(KeyAgreement {
921            crypto: self.crypto,
922            curve: self.curve,
923            private_key: self.private_key,
924            _state: PhantomData,
925        })
926    }
927}
928
929impl KeyAgreement<'_, Ready> {
930    /// Derive the 32-byte raw P-256 shared secret into caller storage.
931    /// Output needs at least 32 bytes; the peer is a 65-byte SEC1 uncompressed
932    /// point validated by the kernel. Return the produced length or a kernel
933    /// error. Inputs and output must not overlap at the ABI boundary.
934    /// The ready object can perform another agreement; this is not a KDF.
935    pub fn generate_secret(
936        &mut self,
937        peer_public_key: &[u8],
938        output: &mut [u8],
939    ) -> Result<usize, CryptoError> {
940        if output.len() < P256_SHARED_SECRET_SIZE {
941            return Err(CryptoError::InvalidOutputLength);
942        }
943        self.crypto
944            .ecdh_do_final(self.curve, &self.private_key, peer_public_key, output)
945    }
946
947    /// Perform ECDH, then HKDF-SHA256 into caller storage; return its byte count.
948    /// Uses a temporary 32-byte shared-secret array and two synchronous syscalls.
949    /// The temporary is filled with zero after HKDF returns, but this is not a
950    /// guaranteed volatile erasure. Salt/info are borrowed for this call.
951    /// Errors from either step propagate; do not consume output on failure.
952    pub fn derive_hkdf_sha256(
953        &mut self,
954        peer_public_key: &[u8],
955        salt: &[u8],
956        info: &[u8],
957        output: &mut [u8],
958    ) -> Result<usize, CryptoError> {
959        let mut shared_secret = [0u8; P256_SHARED_SECRET_SIZE];
960        let shared_len = self.generate_secret(peer_public_key, &mut shared_secret)?;
961        let result = self
962            .crypto
963            .hkdf_sha256(&shared_secret[..shared_len], salt, info, output);
964        shared_secret.fill(0);
965        result
966    }
967}
