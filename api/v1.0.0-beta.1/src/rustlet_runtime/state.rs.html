1//! Separate durable and selection-scoped state, encoded independently.
2use crate::ApduStatus;
3
4/// Typed data owned by the runtime for the lifetime of a selection.
5/// Only `persistent` is written to the registry. Session snapshots stay in RAM.
6pub struct State<P, S = ()> {
7    /// Transaction-owned data serialized into the registry.
8    pub persistent: P,
9    /// Volatile data retained until deselection, reset or terminal failure.
10    pub session: S,
11}
12
13#[doc(hidden)]
14pub trait PostcardState: serde::Serialize + for<'de> serde::Deserialize<'de> + Sized {
15    fn load_postcard_state(&mut self, state: &[u8]) -> Result<(), ApduStatus> {
16        *self = postcard::from_bytes(state).map_err(|_| ApduStatus::wrong_data())?;
17        Ok(())
18    }
19
20    fn save_postcard_state(&self, out: &mut [u8]) -> Result<usize, ApduStatus> {
21        let encoded = postcard::to_slice(self, out).map_err(|_| ApduStatus::wrong_length())?;
22        Ok(encoded.len())
23    }
24}
25
26impl<T> PostcardState for T where T: serde::Serialize + for<'de> serde::Deserialize<'de> + Sized {}
27
28#[cfg(any(feature = "runtime", test))]
29impl<P, S: PostcardState> State<P, S> {
30    pub(crate) fn save_session(&self, out: &mut [u8]) -> Result<usize, ApduStatus> {
31        self.session.save_postcard_state(out)
32    }
33
34    pub(crate) fn load_session(&mut self, bytes: &[u8]) -> Result<(), ApduStatus> {
35        let (session, remaining) =
36            postcard::take_from_bytes(bytes).map_err(|_| ApduStatus::wrong_data())?;
37        if !remaining.is_empty() {
38            return Err(ApduStatus::wrong_data());
39        }
40        self.session = session;
41        Ok(())
42    }
43}

45#[cfg(test)]
46mod tests {
47    use super::*;
48    #[derive(Default, serde::Serialize, serde::Deserialize, PartialEq, Debug)]
49    struct Persistent {
50        counter: u8,
51    }

53    struct Session {
54        secret: [u8; 16],
55        counter: u32,

58    #[test]
59    fn registry_and_session_snapshots_are_separate() {
60        let original = State {
61            persistent: Persistent { counter: 7 },
62            session: Session {
63                secret: [0xA5; 16],
64                counter: 123456,
65            },
66        };
67        let mut registry = [0; 244];
68        let n = original
69            .persistent
70            .save_postcard_state(&mut registry)
71            .map_err(|status| status.to_word())
72            .unwrap();
73        assert_eq!(&registry[..n], &[7]);
74        let mut bytes = [0; 244];
75        let n = original
76            .save_session(&mut bytes)
77            .map_err(|status| status.to_word())
78            .unwrap();
79        let mut rebuilt = State {
80            persistent: Persistent { counter: 6 },
81            session: Session::default(),
82        };
83        rebuilt
84            .load_session(&bytes[..n])
85            .map_err(|status| status.to_word())
86            .unwrap();
87        assert_eq!(rebuilt.persistent.counter, 6);
88        assert_eq!(rebuilt.session, original.session);
89    }

91    #[test]
92    fn empty_states_have_a_valid_zero_length_encoding() {
93        let mut state = State {
94            persistent: (),
95            session: (),
96        };
97        assert_eq!(

99                .persistent

105        assert_eq!(
106            state
107                .save_session(&mut [])
108                .map_err(|status| status.to_word())
109                .unwrap(),
110            0

113            .load_session(&[])
114            .map_err(|status| status.to_word())
115            .unwrap();
116        assert_eq!(
117            state
118                .load_session(&[1])

125    #[test]
126    fn session_codec_rejects_overflow_truncation_and_trailing_bytes() {
127        let mut state = State {
128            persistent: (),
129            session: Session {
130                secret: [0x5A; 16],
131                counter: u32::MAX,
132            },
133        };
134        let mut bytes = [0; 244];
135        assert_eq!(
136            state
137                .save_session(&mut bytes[..16])
138                .err()
139                .map(|status| status.to_word()),
140            Some(ApduStatus::wrong_length().to_word())
141        );
142        let n = state
143            .save_session(&mut bytes)
144            .map_err(|status| status.to_word())
145            .unwrap();
146        assert_eq!(
147            state
148                .load_session(&bytes[..n - 1])
149                .err()
150                .map(|status| status.to_word()),
151            Some(ApduStatus::wrong_data().to_word())
152        );
153        assert_eq!(
154            state
155                .load_session(&bytes[..n + 1])
156                .err()
157                .map(|status| status.to_word()),
158            Some(ApduStatus::wrong_data().to_word())
159        );
160        assert_eq!(state.session.counter, u32::MAX);
161        assert_eq!(state.session.secret, [0x5A; 16]);
162    }
