1//! Reusable SCP03 helpers for Rustlet Security Domains.
2//!
3//! [`S16Session`] owns the protocol mechanics for the common AES-128/S16
4//! profile. A Security Domain still decides which commands it claims, which
5//! instance owns the keys, and which management policy it exposes.

13const CHALLENGE_LEN: usize = 16;
14const CRYPTOGRAM_LEN: usize = 16;
15const MAC_LEN: usize = 16;
16const STATE_INITIALIZED: u8 = 1;
17const STATE_AUTHENTICATED: u8 = 2;
18const KEY_USAGE_ENC: u8 = 1;
19const KEY_USAGE_MAC: u8 = 2;
20const KDF_CARD_CRYPTOGRAM: u8 = 0;
21const KDF_HOST_CRYPTOGRAM: u8 = 1;
22const KDF_S_ENC: u8 = 4;
23const KDF_S_MAC: u8 = 6;
24const KDF_S_RMAC: u8 = 7;

26/// Manifest key identity used by the S16 tutorial engine.
27#[derive(Clone, Copy)]
28pub struct S16Keyset {
29    /// Version selected when INITIALIZE UPDATE uses the version wildcard.
30    pub default_version: u8,
31    /// Identifier reported in the INITIALIZE UPDATE response.
32    pub resolved_id: u8,

35/// Serializable, RAM-only SCP03 S16 session state.
36///
37/// The type intentionally keeps key material private. Its methods compute all
38/// replay-state changes locally and publish them only after the complete
39/// operation succeeds.
40#[derive(Default, crate::serde::Serialize, crate::serde::Deserialize)]
41#[serde(crate = "crate::serde")]
42pub struct S16Session {
43    state: u8,
44    security_level_bits: u8,
45    expected_host_cryptogram: [u8; 16],
46    session_enc_key: [u8; 16],
47    session_mac_key: [u8; 16],
48    session_rmac_key: [u8; 16],
49    command_mac_chain: [u8; 16],
50    response_mac_chain: [u8; 16],
51    #[serde(with = "crate::postcard::fixint::le")]
52    command_enc_counter: u32,
53    #[serde(with = "crate::postcard::fixint::le")]
54    response_enc_counter: u32,
55}

57impl S16Session {
58    /// Serialized upper bound used by the runtime rollback snapshot.
59    pub const SERIALIZED_LEN: usize = 122;
60
61    /// Returns whether EXTERNAL AUTHENTICATE completed successfully.
62    pub fn is_open(&self) -> bool {
63        self.state == STATE_AUTHENTICATED
64    }
65
66    /// Returns the negotiated secure-messaging level.
67    pub fn security_level(&self) -> SecurityLevel {
68        SecurityLevel::from_bits(self.security_level_bits)
69    }
70
71    /// Returns the active command MAC length, or zero while closed.
72    pub fn mac_len(&self) -> usize {
73        if self.is_open() {
74            MAC_LEN
75        } else {
76            0
77        }
78    }

80    /// Clears all derived keys, cryptograms, chains and counters.
81    pub fn reset(&mut self) {
82        self.state = 0;
83        self.security_level_bits = 0;
84        self.expected_host_cryptogram.fill(0);
85        self.session_enc_key.fill(0);
86        self.session_mac_key.fill(0);
87        self.session_rmac_key.fill(0);
88        self.command_mac_chain.fill(0);
89        self.response_mac_chain.fill(0);
90        self.command_enc_counter = 0;
91        self.response_enc_counter = 0;
92    }

94    /// Starts SCP03 S16 and returns the 44-byte INITIALIZE UPDATE response.
95    pub fn initialize_update(
96        &mut self,
97        ctx: &mut RustletCtx,
98        command: &InitializeUpdate<'_>,
99        keyset: S16Keyset,
100        out: &mut [u8],
101    ) -> Result<usize, ApduStatus> {
102        self.reset();
103        if command.host_challenge.len() != CHALLENGE_LEN {
104            return Err(ApduStatus::wrong_length());
105        }
106        // GlobalPlatform mandates P2=0. The runtime interprets that identifier
107        // as a wildcard within the selected Security Domain's key registry.
108        if command.key_id != 0 {
109            return Err(ApduStatus::incorrect_p1_p2());
110        }
111        let key_version = if command.key_version == 0 {
112            keyset.default_version
113        } else {
114            command.key_version
115        };
116        let mut card_challenge = [0u8; CHALLENGE_LEN];
117        RandomData::get_instance(RandomAlgorithm::SecureRandom)
118            .and_then(|mut random| random.generate_data(&mut card_challenge))
119            .map_err(crypto_error_to_status)?;
120        let material =
121            derive_material(ctx, command.host_challenge, &card_challenge, key_version, 0)?;
122        let response_len = 12 + CHALLENGE_LEN + CRYPTOGRAM_LEN;
123        if out.len() < response_len {
124            return Err(ApduStatus::wrong_length());
125        }
126        out[..response_len].fill(0);
127        out[10] = key_version;
128        out[11] = keyset.resolved_id;
129        out[12..28].copy_from_slice(&card_challenge);
130        out[28..44].copy_from_slice(&material.card_cryptogram);
131
132        self.state = STATE_INITIALIZED;
133        self.expected_host_cryptogram = material.host_cryptogram;
134        self.session_enc_key = material.session_enc_key;
135        self.session_mac_key = material.session_mac_key;
136        self.session_rmac_key = material.session_rmac_key;
137        Ok(response_len)
138    }

140    /// Verifies EXTERNAL AUTHENTICATE and publishes the open state atomically.
141    pub fn external_authenticate(
142        &mut self,
143        ctx: &mut RustletCtx,
144        command: &ExternalAuthenticate<'_>,
145    ) -> Result<SecurityLevel, ApduStatus> {
146        if self.state != STATE_INITIALIZED {
147            return Err(ApduStatus::conditions_not_satisfied());
148        }
149        let level = command.security_level;
150        if !is_supported_security_level(level) || command.cla != 0x84 || command.p2 != 0 {
151            self.reset();
152            return Err(ApduStatus::incorrect_p1_p2());
153        }
154        if command.authentication_data.len() != CRYPTOGRAM_LEN + MAC_LEN {
155            self.reset();
156            return Err(ApduStatus::wrong_length());
157        }
158        let (host_cryptogram, received_mac) = command.authentication_data.split_at(CRYPTOGRAM_LEN);
159        let header = [command.cla, 0x82, level.bits(), command.p2, 32];
160        let initial_chain = compute_external_authenticate_cmac(
161            ctx,
162            &self.session_mac_key,
163            &header,
164            host_cryptogram,
165        )?;
166        if !constant_time_eq(host_cryptogram, &self.expected_host_cryptogram)
167            || !constant_time_eq(received_mac, &initial_chain)
168        {
169            self.reset();
170            return Err(ApduStatus::authentication_failed());
171        }
172        self.security_level_bits = level.bits();
173        self.command_mac_chain = initial_chain;
174        self.response_mac_chain = initial_chain;
175        self.state = STATE_AUTHENTICATED;
176        Ok(level)
177    }

179    /// Verifies and optionally decrypts one protected command.
180    pub fn unwrap_command(
181        &mut self,
182        ctx: &mut RustletCtx,
183        command: &WrappedCommand<'_>,
184        out: &mut [u8],
185    ) -> Result<usize, ApduStatus> {
186        let level = self.security_level();
187        if !self.is_open() || !uses_command_mac(level) || command.mac.len() != MAC_LEN {
188            return Err(ApduStatus::conditions_not_satisfied());
189        }
190        let next_chain = compute_chained_cmac(
191            ctx,
192            &self.session_mac_key,
193            &self.command_mac_chain,
194            command.authenticated,
195        )?;
196        if !constant_time_eq(command.mac, &next_chain) {
197            return Err(ApduStatus::conditions_not_satisfied());
198        }
199        let mut next_counter = self.command_enc_counter;
200        let plain_len = if uses_command_encryption(level) && !command.data.is_empty() {
201            let iv = next_encryption_iv(ctx, &self.session_enc_key, &mut next_counter, 1)?;
202            decrypt_iso9797_m2(ctx, &self.session_enc_key, &iv, command.data, out)?
203        } else {
204            if command.data.len() > out.len() {
205                return Err(ApduStatus::wrong_length());
206            }
207            out[..command.data.len()].copy_from_slice(command.data);
208            command.data.len()
209        };
210        self.command_mac_chain = next_chain;
211        self.command_enc_counter = next_counter;
212        Ok(plain_len)
213    }

215    /// Optionally encrypts and authenticates one protected response.
216    pub fn wrap_response(
217        &mut self,
218        ctx: &mut RustletCtx,
219        response: &PlainResponse<'_>,
220        out: &mut [u8],
221    ) -> Result<usize, ApduStatus> {
222        let level = self.security_level();
223        if !self.is_open() || !uses_response_mac(level) {
224            return Err(ApduStatus::conditions_not_satisfied());
225        }
226        let data_offset = 4usize;
227        if out.len() < data_offset {
228            return Err(ApduStatus::wrong_length());
229        }
230        let mut next_counter = self.response_enc_counter;
231        let payload_len = if uses_response_encryption(level) && !response.data.is_empty() {
232            let iv = next_encryption_iv(ctx, &self.session_enc_key, &mut next_counter, 2)?;
233            encrypt_iso9797_m2(
234                ctx,
235                &self.session_enc_key,
236                &iv,
237                response.data,
238                &mut out[data_offset..],
239            )?
240        } else {
241            let end = data_offset
242                .checked_add(response.data.len())
243                .ok_or_else(ApduStatus::wrong_length)?;
244            let destination = out
245                .get_mut(data_offset..end)
246                .ok_or_else(ApduStatus::wrong_length)?;
247            destination.copy_from_slice(response.data);
248            response.data.len()
249        };
250        let mac_offset = data_offset
251            .checked_add(payload_len)
252            .ok_or_else(ApduStatus::wrong_length)?;
253        let total_len = mac_offset
254            .checked_add(MAC_LEN)
255            .ok_or_else(ApduStatus::wrong_length)?;
256        if total_len > out.len() {
257            return Err(ApduStatus::wrong_length());
258        }
259        let next_chain = compute_chained_response_cmac(
260            ctx,
261            &self.session_rmac_key,
262            &self.response_mac_chain,
263            &out[data_offset..mac_offset],
264            response.status,
265        )?;
266        out[0..2].copy_from_slice(&(payload_len as u16).to_be_bytes());
267        out[2] = 0;
268        out[3] = MAC_LEN as u8;
269        out[mac_offset..total_len].copy_from_slice(&next_chain);
270        self.response_mac_chain = next_chain;
271        self.response_enc_counter = next_counter;
272        Ok(total_len)
273    }
274}

276struct DerivedMaterial {
277    card_cryptogram: [u8; 16],
278    host_cryptogram: [u8; 16],
279    session_enc_key: [u8; 16],
280    session_mac_key: [u8; 16],
281    session_rmac_key: [u8; 16],
282}
283
284fn derive_material(
285    ctx: &mut RustletCtx,
286    host_challenge: &[u8],
287    card_challenge: &[u8; 16],
288    key_version: u8,
289    key_id: u8,
290) -> Result<DerivedMaterial, ApduStatus> {
291    let mut enc_key = [0u8; 16];
292    let mut mac_key = [0u8; 16];
293    let enc_len = ctx
294        .load_scp03_key_material(key_version, key_id, KEY_USAGE_ENC, &mut enc_key)
295        .map_err(|_| ApduStatus::referenced_data_not_found())?;
296    let mac_len = ctx
297        .load_scp03_key_material(key_version, key_id, KEY_USAGE_MAC, &mut mac_key)
298        .map_err(|_| ApduStatus::referenced_data_not_found())?;
299    if enc_len != 16 || mac_len != 16 {
300        return Err(ApduStatus::referenced_data_not_found());
301    }
302    let mut context = [0u8; 32];
303    context[..16].copy_from_slice(host_challenge);
304    context[16..].copy_from_slice(card_challenge);
305    let session_enc_key = kdf::<16>(ctx, KDF_S_ENC, &context, &enc_key)?;
306    let session_mac_key = kdf::<16>(ctx, KDF_S_MAC, &context, &mac_key)?;
307    let session_rmac_key = kdf::<16>(ctx, KDF_S_RMAC, &context, &mac_key)?;
308    Ok(DerivedMaterial {
309        card_cryptogram: kdf::<16>(ctx, KDF_CARD_CRYPTOGRAM, &context, &session_mac_key)?,
310        host_cryptogram: kdf::<16>(ctx, KDF_HOST_CRYPTOGRAM, &context, &session_mac_key)?,
311        session_enc_key,
312        session_mac_key,
313        session_rmac_key,
314    })
315}

317/// Implements the SCP03 CMAC-based derivation function.
318pub fn kdf<const N: usize>(
319    ctx: &mut RustletCtx,
320    derivation_constant: u8,
321    context: &[u8],
322    key: &[u8; 16],
323) -> Result<[u8; N], ApduStatus> {
324    let mut provider = ctx.crypto();
325    let mut mac = Mac::<Uninitialized>::new(&mut provider)
326        .init(key, MacAlgorithm::AesCmac)
327        .map_err(crypto_error_to_status)?;
328    mac.update(&[0u8; 11]).map_err(crypto_error_to_status)?;
329    mac.update(&[derivation_constant])
330        .map_err(crypto_error_to_status)?;
331    mac.update(&[0]).map_err(crypto_error_to_status)?;
332    mac.update(&((N as u16) * 8).to_be_bytes())
333        .map_err(crypto_error_to_status)?;
334    mac.update(&[1]).map_err(crypto_error_to_status)?;
335    mac.update(context).map_err(crypto_error_to_status)?;
336    let tag = mac.compute().map_err(crypto_error_to_status)?;
337    let mut out = [0u8; N];
338    out.copy_from_slice(&tag.as_ref()[..N]);
339    Ok(out)
340}

342/// Computes a chained command CMAC without mutating the caller's chain.
343pub fn compute_chained_cmac(
344    ctx: &mut RustletCtx,
345    key: &[u8; 16],
346    chain: &[u8; 16],
347    input: &[u8],
348) -> Result<[u8; 16], ApduStatus> {
349    let mut message = [0u8; 256];
350    let len = 16usize
351        .checked_add(input.len())
352        .filter(|len| *len <= message.len())
353        .ok_or_else(ApduStatus::wrong_length)?;
354    message[..16].copy_from_slice(chain);
355    message[16..len].copy_from_slice(input);
356    let mut provider = ctx.crypto();
357    let tag = provider
358        .compute_mac(key, MacAlgorithm::AesCmac, &message[..len])
359        .map_err(crypto_error_to_status)?;
360    let mut out = [0u8; 16];
361    out.copy_from_slice(tag.as_ref());
362    Ok(out)
363}

365/// Computes R-MAC over protected data followed by SW1-SW2.
366pub fn compute_chained_response_cmac(
367    ctx: &mut RustletCtx,
368    key: &[u8; 16],
369    chain: &[u8; 16],
370    data: &[u8],
371    status: ApduStatus,
372) -> Result<[u8; 16], ApduStatus> {
373    let mut provider = ctx.crypto();
374    let mut mac = Mac::<Uninitialized>::new(&mut provider)
375        .init(key, MacAlgorithm::AesCmac)
376        .map_err(crypto_error_to_status)?;
377    mac.update(chain).map_err(crypto_error_to_status)?;
378    mac.update(data).map_err(crypto_error_to_status)?;
379    mac.update(&[status.sw1, status.sw2])
380        .map_err(crypto_error_to_status)?;
381    let tag = mac.compute().map_err(crypto_error_to_status)?;
382    let mut out = [0u8; 16];
383    out.copy_from_slice(tag.as_ref());
384    Ok(out)
385}

387/// Computes the C-MAC protecting EXTERNAL AUTHENTICATE.
388pub fn compute_external_authenticate_cmac(
389    ctx: &mut RustletCtx,
390    key: &[u8; 16],
391    header: &[u8; 5],
392    host_cryptogram: &[u8],
393) -> Result<[u8; 16], ApduStatus> {
394    let mut provider = ctx.crypto();
395    let mut mac = Mac::<Uninitialized>::new(&mut provider)
396        .init(key, MacAlgorithm::AesCmac)
397        .map_err(crypto_error_to_status)?;
398    mac.update(&[0u8; 16]).map_err(crypto_error_to_status)?;
399    mac.update(header).map_err(crypto_error_to_status)?;
400    mac.update(host_cryptogram)
401        .map_err(crypto_error_to_status)?;
402    let tag = mac.compute().map_err(crypto_error_to_status)?;
403    let mut out = [0u8; 16];
404    out.copy_from_slice(tag.as_ref());
405    Ok(out)
406}

408/// Derives an SCP03 CBC IV and advances only the supplied temporary counter.
409pub fn next_encryption_iv(
410    ctx: &mut RustletCtx,
411    key: &[u8; 16],
412    counter: &mut u32,
413    direction: u8,
414) -> Result<[u8; 16], ApduStatus> {
415    advance_counter(counter)?;
416    let mut block = [0u8; 16];
417    block[0] = direction;
418    block[12..].copy_from_slice(&counter.to_be_bytes());
419    let mut provider = ctx.crypto();
420    let cipher = Cipher::<Uninitialized>::new(&mut provider)
421        .init(
422            key,
423            &[0u8; 16],
424            CipherMode::Encrypt,
425            Algorithm::Aes128EcbNoPadding,
426        )
427        .map_err(crypto_error_to_status)?;
428    let mut out = [0u8; 16];
429    let len = cipher
430        .finish(&block, &mut out)
431        .map_err(crypto_error_to_status)?;
432    if len != out.len() {
433        return Err(ApduStatus::conditions_not_satisfied());
434    }
435    Ok(out)
436}

438/// Decrypts and validates one ISO9797-M2 padded payload.
439pub fn decrypt_iso9797_m2(
440    ctx: &mut RustletCtx,
441    key: &[u8; 16],
442    iv: &[u8; 16],
443    input: &[u8],
444    out: &mut [u8],
445) -> Result<usize, ApduStatus> {
446    let mut provider = ctx.crypto();
447    let cipher = Cipher::<Uninitialized>::new(&mut provider)
448        .init(key, iv, CipherMode::Decrypt, Algorithm::Aes128CbcIso9797M2)
449        .map_err(crypto_error_to_status)?;
450    cipher.finish(input, out).map_err(crypto_error_to_status)
451}

468/// Constant-time equality for fixed protocol fields.
469pub fn constant_time_eq(left: &[u8], right: &[u8]) -> bool {
470    if left.len() != right.len() {
471        return false;
472    }
473    left.iter()
474        .zip(right)
475        .fold(0u8, |difference, (left, right)| difference | (left ^ right))
476        == 0
477}

479fn is_supported_security_level(level: SecurityLevel) -> bool {
480    matches!(level.bits(), 0x01 | 0x03 | 0x11 | 0x13 | 0x33)
481}
482
483fn uses_command_mac(level: SecurityLevel) -> bool {
484    level.bits() & SecurityLevel::C_MAC.bits() != 0
485}
486
487fn uses_command_encryption(level: SecurityLevel) -> bool {
488    level.bits() & SecurityLevel::C_ENC.bits() != 0
489}
490
491fn uses_response_mac(level: SecurityLevel) -> bool {
492    level.bits() & SecurityLevel::R_MAC.bits() != 0
493}
494
495fn uses_response_encryption(level: SecurityLevel) -> bool {
496    level.bits() & SecurityLevel::R_ENC.bits() != 0
497}
498
499fn crypto_error_to_status(_error: crate::CryptoError) -> ApduStatus {
500    ApduStatus::conditions_not_satisfied()
501}
502
503fn advance_counter(counter: &mut u32) -> Result<(), ApduStatus> {
504    let next = counter
505        .checked_add(1)
506        .ok_or_else(ApduStatus::conditions_not_satisfied)?;
507    *counter = next;
508    Ok(())
509}
